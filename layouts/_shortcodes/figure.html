{{- /* Figure shortcode with image processing. Usage: {{< figure src="images/photo.jpg" alt="Description" caption="Optional caption" >}} {{< figure loading="eager" src="images/poster.png" alt="..." >}} (image at the top of the page) Parameters: src - path relative to assets/ (required: the build fails if it is missing) alt - alt text (required: the build fails without it; alt="" only for a purely decorative image) caption - figure caption (optional) width - max width in px (optional, default 800) class - additional CSS class (optional) loading - "lazy" (default) or "eager". Use "eager" for an image visible without scrolling (e.g. a poster at the top of a post): it is then loaded right away and with high priority. */ -}} {{- $src := .Get "src" -}} {{- if not (isset .Params "alt") -}} {{- errorf "figure: manca il testo alternativo (alt), usa alt=\"\" solo per un'immagine decorativa (%s)" .Position -}} {{- end -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" | default "" -}} {{- $width := .Get "width" | default 800 -}} {{- $class := .Get "class" | default "" -}} {{- $loading := .Get "loading" | default "lazy" -}} {{- if not (in (slice "lazy" "eager") $loading) -}} {{- errorf "figure: loading deve essere \"lazy\" o \"eager\", non %q (%s)" $loading .Position -}} {{- end -}}
